## 你只需要做 3 步

### 步驟 1：更新 GAS 入面嘅 Code.gs
最快做法——你個 GAS 專案已經有 Code.gs，只要將呢 3 段換咗佢：

**改動 A — banner（檔頭）**：換成本檔頭（已係 v40）

**改動 B — `_sh()`**（搵 `function _sh(n)` 果行，成段換成）：
```js
var _inInit=false;
function _sh(n){
  const ss=_ss();let s=ss.getSheetByName(n);
  if(!s&&!_inInit){
    _inInit=true;
    try{ setup(); }catch(e){ Logger.log("auto-init error: "+e); }
    _inInit=false;
    s=ss.getSheetByName(n);
  }
  if(!s) throw new Error("找不到工作表："+n);
  return s;
}
```

**改動 C — `setup()`**（搵 `function setup(){`，成個換成）：
```js
function setup(){
  const ss=_ss();
  const cfg=ss.getSheetByName(SHEET_CFG)||ss.insertSheet(SHEET_CFG);
  if(cfg.getLastRow()<1) cfg.getRange(1,1,1,2).setValues([["key","value"]]);
  if(getCfg("year")==="") setCfg("year",YEAR);
  BRANCH_KEYS.forEach(b=>{ if(getCfg("passMark_"+b)==="") setCfg("passMark_"+b,80); });
  const acc=ss.getSheetByName(SHEET_ACC)||ss.insertSheet(SHEET_ACC);
  if(acc.getLastRow()<1){
    acc.getRange(1,1,1,5).setValues([["username","password","role","district","email"]]);
    acc.appendRow(["HKIR",DEFAULT_PW,ROLES.REGION,"",""]);
    Object.keys(DIST_CODES).forEach(code=>{const d=DIST_CODES[code];acc.appendRow([code,DEFAULT_PW,ROLES.REVIEWER,d,""]);acc.appendRow([code+"DC",DEFAULT_PW,ROLES.DC,d,""]);});
  }
  const gs=ss.getSheetByName(SHEET_GROUPS)||ss.insertSheet(SHEET_GROUPS);
  if(gs.getLastRow()<1){ gs.getRange(1,1,1,4).setValues([["district","no","sponsor","sections"]]); GROUPS_DATA.forEach(g=>gs.appendRow([g.d,g.no,g.sponsor,g.sections.join(", ")])); }
  const sub=ss.getSheetByName(SHEET_SUB)||ss.insertSheet(SHEET_SUB);
  if(sub.getLastRow()<1) sub.getRange(1,1,1,SUB_COLS.length).setValues([SUB_COLS.slice()]);
  const sess=ss.getSheetByName(SHEET_SESS)||ss.insertSheet(SHEET_SESS);
  if(sess.getLastRow()<1) sess.getRange(1,1,1,5).setValues([["token","username","role","district","expiresAt"]]);
  try{ setupTemplates(); }catch(e){ Logger.log("templates: "+e); }
  const url=ss.getUrl();
  return "★ 初始化完成！試算表："+url;
}
```

> 偷懶嘅話：直接用 `apps-script/Code.gs` 成個覆蓋 GAS 入面嗰個（base64 前端同之前一模一樣，冇改過）。
